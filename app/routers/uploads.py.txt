import uuid

import httpx
from fastapi import APIRouter, Depends, HTTPException, UploadFile

from app.config import settings
from app.core.deps import get_current_user
from app.models.user import User

router = APIRouter(prefix="/uploads", tags=["uploads"])

BUCKET = "product-images"
ALLOWED_TYPES = {"image/jpeg", "image/png", "image/webp", "image/gif"}


@router.post("/product-image")
async def upload_product_image(
    file: UploadFile,
    current_user: User = Depends(get_current_user),
):
    """
    Uploads an image directly to Supabase Storage via its REST API (no
    Supabase SDK needed - just httpx, which is already a dependency).
    Returns the public URL, ready to be saved as a product's image_url.
    """
    if file.content_type not in ALLOWED_TYPES:
        raise HTTPException(status_code=400, detail="Nur JPEG, PNG, WEBP oder GIF erlaubt.")

    contents = await file.read()
    if len(contents) > 5 * 1024 * 1024:  # 5 MB limit
        raise HTTPException(status_code=400, detail="Datei zu groß (max. 5 MB).")

    extension = file.filename.rsplit(".", 1)[-1] if "." in file.filename else "jpg"
    # Path is namespaced per tenant so one seller can never overwrite another's file.
    path = f"{current_user.tenant_id}/{uuid.uuid4()}.{extension}"

    upload_url = f"{settings.supabase_url}/storage/v1/object/{BUCKET}/{path}"

    async with httpx.AsyncClient() as client:
        response = await client.post(
            upload_url,
            content=contents,
            headers={
                "Authorization": f"Bearer {settings.supabase_service_key}",
                "apikey": settings.supabase_service_key,
                "Content-Type": file.content_type,
            },
        )

    if response.status_code not in (200, 201):
        raise HTTPException(status_code=502, detail="Bild-Upload fehlgeschlagen.")

    public_url = f"{settings.supabase_url}/storage/v1/object/public/{BUCKET}/{path}"
    return {"image_url": public_url}
